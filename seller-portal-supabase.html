🗄️ Database Setup Required

Run this SQL in your Supabase SQL Editor to create the required tables:

-- SELLERS TABLE
CREATE TABLE sellers (
  id UUID PRIMARY KEY,
  email TEXT UNIQUE NOT NULL,
  name TEXT,
  phone TEXT,
  business_name TEXT,
  city TEXT,
  category TEXT,
  free_listings_used INT DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- PRODUCTS TABLE
CREATE TABLE products (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  seller_id UUID REFERENCES sellers(id),
  name TEXT NOT NULL,
  price INT NOT NULL,
  stock INT DEFAULT 0,
  category TEXT,
  emoji TEXT DEFAULT '🛍️',
  description TEXT,
  listing_type TEXT DEFAULT 'free',
  status TEXT DEFAULT 'active',
  transaction_id TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- PAYMENTS TABLE
CREATE TABLE payments (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  seller_id UUID REFERENCES sellers(id),
  amount INT DEFAULT 850,
  transaction_id TEXT,
  status TEXT DEFAULT 'pending',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- ENABLE ROW LEVEL SECURITY
ALTER TABLE sellers ENABLE ROW LEVEL SECURITY;
ALTER TABLE products ENABLE ROW LEVEL SECURITY;
ALTER TABLE payments ENABLE ROW LEVEL SECURITY;

-- POLICIES
CREATE POLICY "Sellers can read own data" ON sellers FOR SELECT USING (auth.uid() = id);
CREATE POLICY "Sellers can insert own data" ON sellers FOR INSERT WITH CHECK (auth.uid() = id);
CREATE POLICY "Sellers can update own data" ON sellers FOR UPDATE USING (auth.uid() = id);
CREATE POLICY "Sellers can read own products" ON products FOR SELECT USING (auth.uid() = seller_id);
CREATE POLICY "Sellers can insert own products" ON products FOR INSERT WITH CHECK (auth.uid() = seller_id);
CREATE POLICY "Sellers can delete own products" ON products FOR DELETE USING (auth.uid() = seller_id);
CREATE POLICY "Sellers can read own payments" ON payments FOR SELECT USING (auth.uid() = seller_id);
CREATE POLICY "Sellers can insert own payments" ON payments FOR INSERT WITH CHECK (auth.uid() = seller_id);
